/** * TMLF Learning Center · Evaluación I-90 vs I-751 * Backend en Google Apps Script + Google Sheets * * - doPost → guarda cada resultado (una fila por intento; evita duplicados por ID) * - doGet → action=check (confirma que se guardó) · action=list (resultados, requiere PIN) * * Instalación: ver instrucciones al final de este archivo. */ const SHEET_NAME = 'Resultados_I90_I751'; const ADMIN_PIN = 'CAMBIA-ESTE-PIN'; // ← cámbialo antes de implementar const TZ = 'America/Mexico_City'; const HEADERS = [ 'Fecha (CDMX)', 'ID', 'Evaluación', 'Nombre completo', 'Clave nombre', 'Intento', 'Inicio', 'Fin', 'Duración (min)', 'Teórica aciertos', 'Teórica total', 'Teórica %', 'Práctica aciertos', 'Práctica total', 'Práctica %', 'Aciertos', 'Total reactivos', 'Total %', 'Aprobado', 'Detalle formas', 'Errores', 'Respuestas (JSON)', 'Versión' ]; const KEYS = [ 'fecha', 'id', 'evalId', 'nombre', 'clave', 'intento', 'inicio', 'fin', 'duracionMin', 'tAciertos', 'tTotal', 'tPct', 'pAciertos', 'pTotal', 'pPct', 'aciertos', 'total', 'pct', 'aprobado', 'formas', 'errores', 'resp', 'version' ]; function sheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(HEADERS); sh.setFrozenRows(1); sh.getRange(1, 1, 1, HEADERS.length) .setFontWeight('bold').setBackground('#0f2a5c').setFontColor('#ffffff'); sh.getRange('G:H').setNumberFormat('@'); // ISO como texto } return sh; } // Evita que un texto se interprete como fórmula en Sheets function safe_(v) { if (typeof v !== 'string') return v; return /^[=+\-@]/.test(v) ? "'" + v : v; } function out_(obj, cb) { const txt = JSON.stringify(obj); if (cb && /^[\w$.]+$/.test(cb)) { return ContentService.createTextOutput(cb + '(' + txt + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(txt).setMimeType(ContentService.MimeType.JSON); } /* ------------------------- GUARDAR ------------------------- */ function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const body = JSON.parse((e.postData && e.postData.contents) || '{}'); if (body.action !== 'save' || !body.record) return out_({ ok: false, error: 'bad request' }); const r = body.record; if (!r.id || !r.nombre) return out_({ ok: false, error: 'incomplete' }); const sh = sheet_(); const last = sh.getLastRow(); let intento = 1; if (last > 1) { const vals = sh.getRange(2, 2, last - 1, 4).getValues(); // ID, Evaluación, Nombre, Clave for (const v of vals) { if (String(v[0]) === String(r.id)) return out_({ ok: true, dup: true }); if (String(v[3]) === String(r.clave)) intento++; } } const fin = new Date(r.fin); sh.appendRow([ Utilities.formatDate(isNaN(fin) ? new Date() : fin, TZ, 'yyyy-MM-dd HH:mm'), r.id, r.evalId, safe_(r.nombre), r.clave, intento, r.inicio, r.fin, r.duracionMin, r.tAciertos, r.tTotal, r.tPct, r.pAciertos, r.pTotal, r.pPct, r.aciertos, r.total, r.pct, r.aprobado ? 'Sí' : 'No', safe_(r.formas), safe_(r.errores), JSON.stringify(r.resp || {}), r.version ]); return out_({ ok: true, intento: intento }); } catch (err) { return out_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } /* ------------------------- LEER ------------------------- */ function doGet(e) { const p = (e && e.parameter) || {}; let res; try { if (p.action === 'check') { res = { ok: true, exists: exists_(p.id) }; } else if (p.action === 'list') { res = String(p.pin) === ADMIN_PIN ? { ok: true, rows: list_() } : { ok: false, error: 'pin' }; } else { res = { ok: true, service: 'TMLF Evaluación I-90 vs I-751' }; } } catch (err) { res = { ok: false, error: String(err) }; } return out_(res, p.callback); } function exists_(id) { if (!id) return false; const sh = sheet_(), last = sh.getLastRow(); if (last < 2) return false; return sh.getRange(2, 2, last - 1, 1).getValues().some(v => String(v[0]) === String(id)); } function list_() { const sh = sheet_(), last = sh.getLastRow(); if (last < 2) return []; const vals = sh.getRange(2, 1, last - 1, KEYS.length).getValues(); return vals.map(row => { const o = {}; KEYS.forEach((k, i) => { let v = row[i]; if (v instanceof Date) v = v.toISOString(); if (typeof v === 'string' && v.charAt(0) === "'") v = v.slice(1); o[k] = v; }); try { o.resp = JSON.parse(o.resp); } catch (err) { o.resp = null; } o.aprobado = o.aprobado === 'Sí' || o.aprobado === true; return o; }).filter(o => o.id); } /* ======================= INSTALACIÓN (5 minutos) ======================= 1.
